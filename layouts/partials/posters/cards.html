{{- /* Film pages listed by a Posters gallery: the films of films.json that match the page's filters (collection, category, size, genre, source, country, letter, year) and, if given, the month. Call with: partial "posters/cards.html" (dict "page" . "month" 0) */ -}} {{- $page := .page -}} {{- $month := .month -}} {{- $retro := eq ($page.Params.collection | default "printed") "retro" -}} {{- $filtro := dict "_" true -}} {{- with $page.Params.source }}{{ $filtro = index (partialCached "posters/fonti.html" . "fonti") (lower .) | default (dict "_" true) }}{{ end -}} {{- with $page.Params.country }}{{ $filtro = index (partialCached "posters/paesi.html" . "paesi") (lower .) | default (dict "_" true) }}{{ end -}} {{- $cards := slice -}} {{- $titoli := partialCached "posters/titoli.html" . "titoli" -}} {{- range hugo.Data.posters.films.films -}} {{- $film := . -}} {{- $images := where .posters "is_retro" $retro -}} {{- with $page.Params.category }}{{ $images = where $images "category" . }}{{ end -}} {{- with $page.Params.size }}{{ $images = where $images "size" . }}{{ end -}} {{- $matches := gt (len $images) 0 -}} {{- with $page.Params.genre }}{{ $matches = and $matches (in $film.genre_slugs .) }}{{ end -}} {{- with $page.Params.source }}{{ $matches = and $matches (index $filtro $film.film_id) }}{{ end -}} {{- with $page.Params.country }}{{ $matches = and $matches (index $filtro $film.film_id) }}{{ end -}} {{- with $page.Params.letter }}{{ $matches = and $matches (eq (index $titoli $film.film_id).lettera .) }}{{ end -}} {{- with $page.Params.year -}} {{- $screened := false -}} {{- range $film.screenings -}} {{- if eq (int .year) (int $page.Params.year) -}} {{- if or (not $month) (eq (int .month) (int $month)) }}{{ $screened = true }}{{ end -}} {{- end -}} {{- end -}} {{- $matches = and $matches $screened -}} {{- end -}} {{- if $matches -}} {{- $filmPage := site.GetPage (printf "/posters/films/%s" $film.film_id) -}} {{- if not $filmPage }}{{ errorf "Poster film page missing: %s" $film.film_id }}{{ end -}} {{- $cards = $cards | append (dict "k" (index $titoli $film.film_id).chiave "p" $filmPage) -}} {{- end -}} {{- end -}} {{- /* In title order (display titles). */ -}} {{- $ordinate := slice -}} {{- range sort $cards "k" }}{{ $ordinate = $ordinate | append .p }}{{ end -}} {{- return $ordinate -}}
