$count }} {{- $v := add $best.lmin (mul (float $i) $best.step) }} {{- if and (ge $v (sub $dmin 1e-9)) (le $v (add $dmax 1e-9)) }}{{ $out = $out | append $v }}{{ end }} {{- end }} {{- return $out }}
