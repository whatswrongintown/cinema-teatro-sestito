{{- /* Vertical bar chart: posters per screening year, as inline SVG. Data: static/data/posters_year_counts.csv (year, n_posters) and static/data/posters_no_year.csv (n_posters without a screening date). Every year from the first to the last is shown, with 0 where there is no poster. */ -}} {{- $rows := resources.Get "data/posters_year_counts.csv" | transform.Unmarshal }} {{- $noYear := 0 }} {{- range after 1 (resources.Get "data/posters_no_year.csv" | transform.Unmarshal) }}{{ $noYear = int (index . 0) }}{{ end }} {{- /* Year -> count, then the full year range */ -}} {{- $byYear := dict }} {{- $first := 9999 }}{{ $last := 0 }}{{ $total := 0 }}{{ $max := 0 }}{{ $peak := 0 }} {{- range after 1 $rows }} {{- $y := int (index . 0) }}{{ $n := int (index . 1) }} {{- $byYear = merge $byYear (dict (string $y) $n) }} {{- $first = math.Min $first $y }}{{ $last = math.Max $last $y }} {{- $total = add $total $n }} {{- if gt $n $max }}{{ $max = $n }}{{ $peak = $y }}{{ end }} {{- end }} {{- $first = int $first }}{{ $last = int $last }} {{- $years := seq $first $last }} {{- /* Y axis: a step giving at most 5 intervals */ -}} {{- $step := 500 }} {{- range slice 500 200 100 50 40 25 20 10 5 2 1 }} {{- if le (math.Ceil (div (float $max) (float .))) 5.0 }}{{ $step = . }}{{ end }} {{- end }} {{- $top := mul $step (int (math.Ceil (div (float $max) (float $step)))) }} {{- if eq $top 0 }}{{ $top = $step }}{{ end }} {{- /* Geometry (SVG user units) */ -}} {{- $W := 640.0 }}{{ $H := 340.0 }} {{- $ml := 48.0 }}{{ $mr := 10.0 }}{{ $mt := 12.0 }}{{ $mb := 56.0 }} {{- $pw := sub $W (add $ml $mr) }}{{ $ph := sub $H (add $mt $mb) }} {{- $band := div $pw (float (len $years)) }} {{- $bw := mul $band 0.72 }} {{- $base := add $mt $ph }} {{- $desc := printf "Grafico a barre verticali: numero di manifesti e locandine per anno di proiezione del film nel cinema, dal %d al %d. Sono conteggiati i manifesti con una data di proiezione nota (%s in tutto); gli anni senza manifesti hanno valore zero. Il massimo è nel %d, con %s manifesti." $first $last (lang.FormatNumber 0 $total) $peak (lang.FormatNumber 0 $max) }}
