{{- /* Image gallery. gallery_dir is either: - relative ("galleries/"): a folder inside the page bundle, read as page resources; - absolute ("/images/"): a folder under static/images/ (historical sources). The galleryDescriptions front matter key is the gallery_dir value as written (a leading slash is optional). */ -}} {{- $dir := .Get "gallery_dir" -}} {{- $galleryDescriptions := .Page.Params.galleryDescriptions -}} {{- $supportedExtensions := slice ".jpg" ".jpeg" ".png" ".gif" ".webp" ".tif" ".tiff" ".bmp" -}} {{- $items := slice -}} {{- $ok := true -}} {{- if not $dir -}} {{- errorf "image-gallery shortcode in %q: missing required gallery_dir argument" .Page.File.Path -}} {{- $ok = false -}} {{- else if or (strings.Contains $dir "..") (strings.Contains $dir "\\") -}} {{- errorf "image-gallery shortcode in %q: gallery_dir %q must not contain parent-directory segments" .Page.File.Path $dir -}} {{- $ok = false -}} {{- else if strings.HasPrefix $dir "/" -}} {{- /* Static mode: /images/ under static/ */ -}} {{- if not (strings.HasPrefix $dir "/images/") -}} {{- errorf "image-gallery shortcode in %q: absolute gallery_dir %q must be under /images" .Page.File.Path $dir -}} {{- $ok = false -}} {{- else if not (fileExists (printf "/static%s" $dir)) -}} {{- errorf "image-gallery shortcode in %q: gallery directory %q does not exist under static/" .Page.File.Path $dir -}} {{- $ok = false -}} {{- else -}} {{- range readDir (printf "/static%s" $dir) -}} {{- if .IsDir -}} {{- errorf "image-gallery shortcode in %q: nested directory %q in gallery %q is not supported" $.Page.File.Path .Name $dir -}} {{- $ok = false -}} {{- else if not (in $supportedExtensions (path.Ext .Name | strings.ToLower)) -}} {{- errorf "image-gallery shortcode in %q: unsupported file %q in gallery %q" $.Page.File.Path .Name $dir -}} {{- $ok = false -}} {{- else -}} {{- $image := resources.Get (printf "%s/%s" $dir .Name) -}} {{- if $image -}} {{- $items = $items | append (dict "name" .Name "image" $image) -}} {{- else -}} {{- errorf "image-gallery shortcode in %q: unable to load image %q from gallery %q" $.Page.File.Path .Name $dir -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Bundle mode: folder inside the page bundle */ -}} {{- $dir = strings.TrimSuffix "/" $dir -}} {{- range .Page.Resources.Match (printf "%s/*" $dir) -}} {{- $name := path.Base .Name -}} {{- if not (in $supportedExtensions (path.Ext $name | strings.ToLower)) -}} {{- errorf "image-gallery shortcode in %q: unsupported file %q in gallery %q" $.Page.File.Path $name $dir -}} {{- $ok = false -}} {{- else -}} {{- $items = $items | append (dict "name" $name "image" .) -}} {{- end -}} {{- end -}} {{- if not $items -}} {{- errorf "image-gallery shortcode in %q: gallery directory %q is missing or has no images in the page bundle" .Page.File.Path $dir -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- if and $ok $items -}} {{- $pageAlt := "" -}} {{- $imageDescriptions := dict -}} {{- with $galleryDescriptions -}} {{- $description := index . $dir -}} {{- if not $description -}} {{- $description = index . (printf "/%s" (strings.TrimPrefix "/" $dir)) -}} {{- end -}} {{- with $description -}} {{- $pageAlt = .pageAlt | default "" -}} {{- $imageDescriptions = .images | default dict -}} {{- end -}} {{- end -}} {{- if and (not $pageAlt) (eq (len $imageDescriptions) 0) -}} {{- warnf "image-gallery shortcode in %q: gallery %q has no galleryDescriptions metadata; using filename fallback" .Page.File.Path $dir -}} {{- end -}}
